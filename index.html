# -*- coding: utf-8 -*- """时间线视图：横向甘特图（Notion 风格） 左侧任务树（父任务可展开子条），右侧日期轴横条： 创建日 → 截止日(未完成) / 完成日(已完成) 颜色语义与日历一致: 蓝=进行中, 红=逾期未完成, 绿=已完成 """ from datetime import date, timedelta from PySide6.QtCore int(h[2:4], 16), int(h[4:6], 16) return f"rgba({r}, {g}, {b}, {alpha})" def task_color(task): """颜色语义: 绿=已完成, 红=逾期未完成, 蓝=进行中""" if task["completed"]: return st.SUCCESS end = task.get("end") if end: try: if date.fromisoformat(end) < date.today(): return st.DANGER except ValueError: pass return st.PRIMARY class GanttCanvas(QWidget): """甘特图画布：完整绘制任务树 + 日期轴""" task_activated = Signal(int) def self.setMouseTracking(True) self.setToolTipDuration(-1) # ---------- 数据 ---------- def set_data(self, parents, collapsed, search=""): self.collapsed = collapsed self.rows = [] for p in parents: in ("start", "end"): v = t.get(key) if v: try: dates.append(date.fromisoformat(v)) except ValueError: pass # 日期轴范围: 从结束端截断, 保证"今天"始终可见 latest = max(dates) earliest = min(dates) start_limit = latest - t["sub_count"] if frac > 0 and not done: p.setBrush(QColor(255, 255, 255, 160)) p.setPen(Qt.NoPen) p.drawRoundedRect(QRectF(x1, by, (x2 - x1) * frac, bar_h), bar_h / 2, bar_h / 2) p.end() # ----------
